// 中文用途：读冻结的原图轮廓输入，运行取证副本并输出分布/反事实；结果不写回生产检测器。
#include "config/config.hpp"
#include "corners/corner_edge_fit.hpp"
#include <fstream>
#include <sstream>
#include <iostream>
#include <iomanip>
#include <map>
#include <algorithm>
namespace mark{extern std::string probe_mode;extern std::map<std::string,std::vector<double>> probe_values;}
int main(int argc,char**argv){
 auto cfg=mark::loadConfig(argv[1]).detector_config.corner_;mark::probe_mode=argv[3];
 std::ifstream f(argv[2]);std::string line;std::cout<<std::setprecision(17);
 while(std::getline(f,line)){
 std::istringstream in(line);int fid,h,p,cid,n;in>>fid>>h>>p>>cid>>n;
 std::array<std::array<cv::Point2d,2>,2> edges;for(auto &e:edges)for(auto &q:e)in>>q.x>>q.y;
 std::vector<cv::Point> ring(n);for(auto &q:ring)in>>q.x>>q.y;
 bool orig;in>>orig;double ox=0,oy=0;if(orig)in>>ox>>oy;std::string reason;in>>std::quoted(reason);
 mark::probe_values.clear();auto r=mark::fitObservedEdgePair(ring,edges,cfg);
 std::cout<<"{\"frame\":"<<fid<<",\"hypothesis\":"<<h<<",\"physical\":"<<p<<",\"component\":"<<cid<<",\"original_success\":"<<orig<<",\"success\":"<<bool(r.evidence)<<",\"mode\":\""<<mark::probe_mode<<"\"";
 if(r.evidence)std::cout<<",\"corner\":["<<r.evidence->intersection_.x<<","<<r.evidence->intersection_.y<<"],\"original_delta\":"<<(orig?cv::norm(r.evidence->intersection_-cv::Point2d(ox,oy)):-1)<<",\"support_mean\":["<<r.evidence->line_mean_residual_px_[0]<<","<<r.evidence->line_mean_residual_px_[1]<<"],\"support_extension\":["<<r.evidence->support_extension_px_[0]<<","<<r.evidence->support_extension_px_[1]<<"]";
 if(r.evidence){ std::cout<<",\"support_arcs\":["; for(int e=0;e<2;++e){if(e)std::cout<<",";std::cout<<"[";bool pf=true;for(auto q:r.evidence->original_support_arcs_[e]){if(!pf)std::cout<<",";pf=false;std::cout<<"["<<q.x<<","<<q.y<<"]";}std::cout<<"]";}std::cout<<"],\"turn_arc\":[";bool pf=true;for(auto q:r.evidence->original_turn_arc_){if(!pf)std::cout<<",";pf=false;std::cout<<"["<<q.x<<","<<q.y<<"]";}std::cout<<"]"; }
 std::cout<<",\"reason\":"<<std::quoted(r.reason)<<",\"metrics\":{";bool first=true;
 for(auto &kv:mark::probe_values){if(!first)std::cout<<",";first=false;auto &v=kv.second;std::sort(v.begin(),v.end());double sum=0;for(double x:v)sum+=x;
 std::cout<<std::quoted(kv.first)<<":{\"n\":"<<v.size()<<",\"min\":"<<v.front()<<",\"p50\":"<<v[v.size()/2]<<",\"p95\":"<<v[std::min(v.size()-1,size_t(v.size()*.95))]<<",\"max\":"<<v.back()<<",\"sum\":"<<sum<<",\"hist\":{";
 std::map<int,size_t>hist;double step=kv.first=="rejected_mean_residual"?.05:1.;for(double x:v)++hist[int(std::floor(x/step))];bool hf=true;for(auto &z:hist){if(!hf)std::cout<<",";hf=false;std::cout<<std::quoted(std::to_string(z.first))<<":"<<z.second;}std::cout<<"},\"hist_step\":"<<step<<"}";
 }
 std::cout<<"}}\n";
 }
}
