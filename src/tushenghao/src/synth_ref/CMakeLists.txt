# S01–S08：来源、几何、真值、编码、生成事务与独立验证分别拥有实现。
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(nlohmann_json 3.2 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(SG_CORE_SOURCES inputs/source_inputs.cpp model/model.cpp enumeration/grid.cpp geometry/transforms.cpp
    perturbation/perturb.cpp raster/raster.cpp scene/cases.cpp truth/truth_record.cpp
    records/record_codec.cpp generation/generation.cpp verification/verify.cpp)
set(SG_CORE_HEADERS model grid transforms perturb raster cases records verify)
set(SG_CODE "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/build_info.hpp.in")
foreach(source IN LISTS SG_CORE_SOURCES)
    list(APPEND SG_CODE "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
endforeach()
foreach(header IN LISTS SG_CORE_HEADERS)
    list(APPEND SG_CODE "${VISION_ROOT}/include/vision/synth_ref/${header}.hpp")
endforeach()
list(SORT SG_CODE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SG_CODE})
string(TOUPPER "${CMAKE_BUILD_TYPE}" SG_BUILD_TYPE)
set(SG_DIGEST "provenance_schema=1\ncompiler=${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}\nbuild=${CMAKE_BUILD_TYPE}\nflags=${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${SG_BUILD_TYPE}}\n")
foreach(file IN LISTS SG_CODE)
    file(SHA256 "${file}" hash)
    file(RELATIVE_PATH relative "${VISION_ROOT}" "${file}")
    string(APPEND SG_DIGEST "${relative} ${hash}\n")
endforeach()
string(SHA256 SG_CODE_HASH "${SG_DIGEST}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/synthetic_gen_core-sources.manifest" "${SG_DIGEST}")
# 原生成器固定版本保持，PROJECT_VERSION 只在此目录作用域用于既有模板。
set(PROJECT_VERSION "1.0.0")
configure_file(build_info.hpp.in generated/vision/synth_ref/build_info.hpp @ONLY)
add_library(synthetic_gen_core ${SG_CORE_SOURCES})
add_library(vision::synth_ref ALIAS synthetic_gen_core)
set_property(TARGET synthetic_gen_core PROPERTY MARK_GENERATED_INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/build_info.hpp.in")
target_include_directories(synthetic_gen_core PUBLIC "${VISION_ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}/generated" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(synthetic_gen_core PUBLIC ${OpenCV_LIBS} nlohmann_json::nlohmann_json OpenSSL::Crypto)
target_compile_features(synthetic_gen_core PUBLIC cxx_std_17)
set_target_properties(synthetic_gen_core PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(synthetic_gen_core PRIVATE -Wall -Wextra -Wpedantic)
