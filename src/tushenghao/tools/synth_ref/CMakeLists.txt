# 独立入口只装配自己的库／CLI／测试；不向上递归导入整个工程。
cmake_minimum_required(VERSION 3.16)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(synthetic_gen VERSION 1.0.0 LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    get_filename_component(VISION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
    include(CTest)
endif()
if(NOT TARGET synthetic_gen_core)
    add_subdirectory("${VISION_ROOT}/src/synth_ref" synthetic_core)
endif()
add_executable(synthetic_gen main.cpp)
target_link_libraries(synthetic_gen PRIVATE vision::synth_ref)
set_target_properties(synthetic_gen PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    add_subdirectory("${VISION_ROOT}/tests/synth_ref" synth_tests)
endif()
# CLI 有自己的来源清单；不将入口／测试文件混入核心库身份。
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" cli_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" cmake_hash)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/synthetic_gen-sources.manifest" "main.cpp ${cli_hash}\nCMakeLists.txt ${cmake_hash}\n")
