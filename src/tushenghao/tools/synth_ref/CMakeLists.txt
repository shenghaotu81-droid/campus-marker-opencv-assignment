# 独立工具工程，依赖只链接本工具目标，不改上层工程或下载第三方代码。
cmake_minimum_required(VERSION 3.16)
project(synthetic_gen VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(nlohmann_json 3.2 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
# B05：生成库身份只覆盖实际核心输入；CLI 与测试各自另有目标 manifest。
set(SG_CORE_SOURCES src/model.cpp src/grid.cpp src/transforms.cpp src/perturb.cpp
    src/raster.cpp src/cases.cpp src/records.cpp src/verify.cpp)
set(SG_CORE_HEADERS model grid transforms perturb raster cases records verify)
set(SG_CODE "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt"
    "${CMAKE_CURRENT_SOURCE_DIR}/include/synthetic_gen/build_info.hpp.in")
foreach(source IN LISTS SG_CORE_SOURCES)
    list(APPEND SG_CODE "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
endforeach()
foreach(header IN LISTS SG_CORE_HEADERS)
    list(APPEND SG_CODE "${CMAKE_CURRENT_SOURCE_DIR}/include/synthetic_gen/${header}.hpp")
endforeach()
list(SORT SG_CODE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SG_CODE})
string(TOUPPER "${CMAKE_BUILD_TYPE}" SG_BUILD_TYPE)
set(SG_DIGEST "provenance_schema=1\ncompiler=${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}\nbuild=${CMAKE_BUILD_TYPE}\nflags=${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${SG_BUILD_TYPE}}\n")
foreach(file IN LISTS SG_CODE)
    file(SHA256 "${file}" hash)
    file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}" "${file}")
    string(APPEND SG_DIGEST "${relative} ${hash}\n")
endforeach()
string(SHA256 SG_CODE_HASH "${SG_DIGEST}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/synthetic_gen_core-sources.manifest" "${SG_DIGEST}")
configure_file(include/synthetic_gen/build_info.hpp.in generated/synthetic_gen/build_info.hpp @ONLY)
add_library(synthetic_gen_core ${SG_CORE_SOURCES})
set_property(TARGET synthetic_gen_core PROPERTY MARK_GENERATED_INPUTS
    "${CMAKE_CURRENT_SOURCE_DIR}/include/synthetic_gen/build_info.hpp.in")
target_include_directories(synthetic_gen_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(synthetic_gen_core PUBLIC ${OpenCV_LIBS} nlohmann_json::nlohmann_json OpenSSL::Crypto)
target_compile_options(synthetic_gen_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(synthetic_gen src/main.cpp)
target_link_libraries(synthetic_gen PRIVATE synthetic_gen_core)
set_target_properties(synthetic_gen PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
# 普通 C++ 测试显式返回失败，Release 不依赖 assert。
include(CTest)
if(BUILD_TESTING)
  add_executable(synthetic_gen_tests tests/tests.cpp)
  target_link_libraries(synthetic_gen_tests PRIVATE synthetic_gen_core)
  target_compile_definitions(synthetic_gen_tests PRIVATE SG_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
  foreach(SG_TEST model transforms grid raster perturb cases workflow)
    add_test(NAME ${SG_TEST} COMMAND synthetic_gen_tests ${SG_TEST})
    set_tests_properties(${SG_TEST} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endforeach()
endif()

foreach(target synthetic_gen synthetic_gen_tests)
    if(TARGET ${target})
        get_target_property(sources ${target} SOURCES)
        set(manifest "${SG_DIGEST}")
        if(target STREQUAL "synthetic_gen" OR target STREQUAL "synthetic_gen_tests")
            list(APPEND sources include/synthetic_gen/core.hpp)
        endif()
        foreach(source IN LISTS sources)
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
            file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" hash)
            string(APPEND manifest "${source} ${hash}\n")
        endforeach()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${target}-sources.manifest" "${manifest}")
    endif()
endforeach()
