cmake_minimum_required(VERSION 3.16)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(block3_fixture LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    get_filename_component(VISION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
    find_package(OpenCV REQUIRED)
    include("${VISION_ROOT}/cmake/provenance.cmake")
    add_subdirectory("${VISION_ROOT}/src/types" modules/types)
    add_subdirectory("${VISION_ROOT}/src/math" modules/math)
    add_subdirectory("${VISION_ROOT}/src/config" modules/config)
    add_subdirectory("${VISION_ROOT}/src/observability" modules/observability)
    add_subdirectory("${VISION_ROOT}/src/detection" modules/detection)
    add_subdirectory("${VISION_ROOT}/src/synth_ref" modules/synth_ref)
    add_subdirectory("${VISION_ROOT}/tests/support" test_support)
endif()
add_executable(block3_fixture generation/main.cpp)
target_link_libraries(block3_fixture PRIVATE synthetic_gen_core)
target_compile_definitions(block3_fixture PRIVATE MODEL_YAML="${CMAKE_CURRENT_SOURCE_DIR}/../../config/marker_geometry.yaml")

# 用途分目录，CMake入口和可执行目标名保持兼容。
# 隔离定位runner仅工具目标连接Detector与真值读取；正式Detector库无反向依赖。
add_executable(block3_measure calibration/measure.cpp)
target_link_libraries(block3_measure PRIVATE mark_detector_internal_headers synthetic_gen_core)
target_compile_definitions(block3_measure PRIVATE DETECTOR_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/../../config/detector.yaml")

add_executable(block3_resources validation/resources.cpp)
target_link_libraries(block3_resources PRIVATE mark_detector_internal_headers mark_test_support)
add_executable(block3_verify validation/verify.cpp)
target_link_libraries(block3_verify PRIVATE mark_detector_internal_headers synthetic_gen_core)
target_compile_definitions(block3_verify PRIVATE DETECTOR_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/../../config/detector.yaml")

# 实拍拒绝原因只读诊断；仍复用Detector，不改变生产预算或读取真值。
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(block3_video_diagnose diagnostics/video/video_diagnose.cpp)
target_link_libraries(block3_video_diagnose PRIVATE mark_detector_internal_headers nlohmann_json::nlohmann_json)
add_executable(block3_video_corner_measure diagnostics/corners/video_corner_measure.cpp)
target_link_libraries(block3_video_corner_measure PRIVATE mark_detector_internal_headers nlohmann_json::nlohmann_json)
add_executable(block3_parent_diagnose diagnostics/parent/parent_diagnose.cpp)
target_include_directories(block3_parent_diagnose PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(block3_parent_diagnose PRIVATE mark_detector_internal_headers nlohmann_json::nlohmann_json OpenSSL::Crypto)


foreach(target block3_video_diagnose block3_video_corner_measure block3_parent_diagnose block3_measure)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endforeach()

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    foreach(target block3_fixture block3_measure block3_resources block3_verify block3_video_diagnose block3_video_corner_measure block3_parent_diagnose)
        mark_target_provenance(${target} "${VISION_ROOT}")
        if(target STREQUAL "block3_measure" OR target STREQUAL "block3_verify")
            get_target_property(hash ${target} MARK_SOURCE_HASH)
            target_compile_definitions(${target} PRIVATE MEASURE_CODE_HASH="${hash}")
        endif()
    endforeach()
endif()
