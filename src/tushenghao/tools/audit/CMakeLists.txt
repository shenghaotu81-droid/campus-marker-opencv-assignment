find_package(OpenSSL REQUIRED COMPONENTS Crypto)
foreach(name geometry decode temporal)
    add_executable(${name}_audit ${name}/${name}_audit.cpp)
    target_link_libraries(${name}_audit PRIVATE vision_app_internal_headers)
    target_include_directories(${name}_audit PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
    target_compile_definitions(${name}_audit PRIVATE MARK_COMMIT="${MARK_COMMIT}")
    add_dependencies(${name}_audit marker_display_probe)
endforeach()
target_link_libraries(decode_audit PRIVATE OpenSSL::Crypto)
target_link_libraries(temporal_audit PRIVATE OpenSSL::Crypto)
target_compile_definitions(decode_audit PRIVATE MARK_DEFAULT_CONFIG="${VISION_ROOT}/config/detector.yaml")
