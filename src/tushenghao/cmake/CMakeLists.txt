# 全模块装配结束后登记实际目标闭包；不把源表放回顶层。
function(vision_targets_in directory output)
    get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(child IN LISTS children)
        vision_targets_in("${child}" nested)
        list(APPEND targets ${nested})
    endforeach()
    set(${output} "${targets}" PARENT_SCOPE)
endfunction()
vision_targets_in("${VISION_ROOT}" all_targets)
foreach(target IN LISTS all_targets)
    get_target_property(type "${target}" TYPE)
    if(NOT type STREQUAL "INTERFACE_LIBRARY" AND NOT target STREQUAL "synthetic_gen_core")
        mark_target_provenance(${target} "${VISION_ROOT}")
        get_target_property(hash "${target}" MARK_SOURCE_HASH)
        if(target STREQUAL "vision_app")
            target_compile_definitions(${target} PRIVATE MARK_CODE_HASH="${hash}" MARK_COMMIT="${MARK_COMMIT}" MARK_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
        elseif(target MATCHES "^(marker_app|decode_audit|geometry_audit|temporal_audit)$")
            target_compile_definitions(${target} PRIVATE MARK_CODE_HASH="${hash}")
        elseif(target MATCHES "^block3_(measure|verify)$")
            target_compile_definitions(${target} PRIVATE MEASURE_CODE_HASH="${hash}")
        endif()
    endif()
endforeach()
